from fastapi import FastAPI, Query, HTTPException
import httpx
import time

app = FastAPI(
    title="CTX Product API",
    version="1.0.0"
)

CTX_EBOOK_URL = (
    "https://ctx.cretec.kr/CtxApp/ebook/"
    "selectEbookUninumSearch.do"
)

CTX_POWER_URL = (
    "https://ctx.cretec.kr/CtxApp/ctx/"
    "selectPowerSearchJson.do"
)


HEADERS = {
    "User-Agent": (
        "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
        "AppleWebKit/537.36 (KHTML, like Gecko) "
        "Chrome/151.0.0.0 Safari/537.36"
    ),
    "Accept": "application/json, text/javascript, */*; q=0.01",
    "Referer": "https://ctx.cretec.kr/"
}


@app.get("/ctx/product")
async def get_product(
    prodCd: str = Query(..., description="CTX 상품번호")
):

    timestamp = str(int(time.time() * 1000))

    # -----------------------------------------
    # 1. Ebook API
    # -----------------------------------------

    ebook_params = {
        "prodCd": prodCd,
        "itemCd": "",
        "_": timestamp
    }

    # -----------------------------------------
    # 2. Power Search API
    # -----------------------------------------

    power_params = {
        "prod_cd": prodCd,
        "keyword": "",
        "_": timestamp
    }

    try:

        async with httpx.AsyncClient(
            timeout=20.0,
            follow_redirects=True
        ) as client:

            # 두 API 동시 호출
            ebook_response, power_response = await __import__(
                "asyncio"
            ).gather(

                client.get(
                    CTX_EBOOK_URL,
                    params=ebook_params,
                    headers=HEADERS
                ),

                client.get(
                    CTX_POWER_URL,
                    params=power_params,
                    headers=HEADERS
                )
            )

            ebook_response.raise_for_status()
            power_response.raise_for_status()

            ebook_data = ebook_response.json()
            power_data = power_response.json()

        # -----------------------------------------
        # Ebook 데이터
        # -----------------------------------------

        ebook_result = ebook_data.get("result", [])

        if not ebook_result:
            raise HTTPException(
                status_code=404,
                detail=f"Ebook 상품을 찾을 수 없습니다: {prodCd}"
            )

        ebook_item = ebook_result[0]

        ebook_fields = [
            "procode",
            "proino",
            "procpn",
            "cateNm",
            "proinm",
            "prostd",
            "promod",
            "pconam",
            "linkX1",
            "linkY1",
            "linkX2",
            "linkY2",
            "mastDb",
            "folder",
            "pg"
        ]

        ebook_result_data = {
            key: ebook_item.get(key)
            for key in ebook_fields
        }

        # -----------------------------------------
        # Power Search 데이터
        # -----------------------------------------

        power_result = power_data.get("result", {})
        power_items = power_result.get("items", [])

        if not power_items:
            raise HTTPException(
                status_code=404,
                detail=f"Power 상품을 찾을 수 없습니다: {prodCd}"
            )

        power_item = power_items[0]

        # qties 분리
        # 예: "1,0,0,EA,"
        #     [0] = 1
        #     [1] = 0
        #     [2] = 0
        #     [3] = EA
        qties = (power_item.get("qties") or "").split(",")

        moq = qties[0] if len(qties) > 0 else ""
        unit = qties[3] if len(qties) > 3 else ""


        power_fields = [

            "qties",
            "std_sale_pric",
            "std_unit_pric",
            "sale_prices",
            "std_unit_pric_rte",

            "item_stdz",
            "img_file",
            "image_name",

            "d1sum",
            "s1sum",
            "order_qty",

            "ebook_yn",
            "item_use_cd",

            "stpay",
            "recp_expt_dt",
            "recp_expt_dt_s1",

            "dept_info",
            "dept_info_s1",

            "ritteryn",
            "imp_cargo_serv",
            "cg_psvc_yn",
            "no_sale_item_yn",

            "modl_no",
            "yusa",
            "brnd_cd",

            "rtn_imps_yn",
            "imp_delv_serv",
            "imp_delv_serv_all",

            "no_air_trans_yn",
            "hand_with_care_yn",
            "delv_pay_item_cls_cd",

            "online_no_sale",
            "direct_no_sale_yn",

            "chemi_yn",
            "as_cost_yn",
            "oil_remove_yn",
            "indi_cargo_yn",
            "rcmd_cls"
        ]


        power_result_data = {
            key: power_item.get(key)
            for key in power_fields
        }


        # qties에서 추출한 값 추가
        power_result_data["moq"] = moq
        power_result_data["unit"] = unit

        # -----------------------------------------
        # 최종 결과
        # -----------------------------------------

        return {
            **ebook_result_data,
            **power_result_data
        }

    except httpx.TimeoutException:

        raise HTTPException(
            status_code=504,
            detail="CTX 서버 응답 시간 초과"
        )

    except httpx.HTTPStatusError as e:

        raise HTTPException(
            status_code=e.response.status_code,
            detail=f"CTX 서버 HTTP 오류: {e.response.status_code}"
        )

    except HTTPException:
        raise

    except Exception as e:

        raise HTTPException(
            status_code=500,
            detail=f"데이터 수집 오류: {str(e)}"
        )
